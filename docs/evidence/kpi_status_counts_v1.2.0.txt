# SQL Query: Status counts per kpi_id x basis x status (policy_version = 'v1.2.0')

FROM kpi_values
WHERE policy_version = 'v1.2.0'
GROUP BY kpi_id, basis, status
ORDER BY kpi_id, basis, status;
